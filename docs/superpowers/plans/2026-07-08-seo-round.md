```python
import pytest
from pathlib import Path

FIXTURES = Path(__file__).parent / "data" / "synthetic"

CASES = [
    ("us-checking.pdf", ["Date", "Description", "Withdrawals", "Deposits", "Balance"], "06/03/2026"),
    ("us-card.pdf", ["Date", "Description", "Amount"], "06/03"),
    ("uk-current.pdf", ["Date", "Description", "Money out", "Money in", "Balance"], "3 Jun 2026"),
    ("eu-giro.pdf", ["Datum", "Omschrijving", "Af", "Bij", "Saldo"], "03-06-2026"),
]

@pytest.mark.parametrize("fixture,expected_headers,sample_date", CASES)
def test_western_format_replicated(fixture, expected_headers, sample_date):
    from parsers.layout_replica_parser import create_layout_replica_parser
    parser = create_layout_replica_parser()          # match worker.py's construction exactly
    parser.parse(str(FIXTURES / fixture))            # match worker.py's parse call exactly
    headers = [c.header for c in parser.table_columns]
    assert headers == expected_headers
    assert len(parser.table_rows) == 8
    assert any((row.cells[0] or "").strip().startswith(sample_date) for row in parser.table_rows)
```
